# Math core library and GoogleTest suite. Usable standalone:
#   cmake -S Math -B build/Math -DCMAKE_BUILD_TYPE=Debug
cmake_minimum_required(VERSION 3.20)
project(MathStandalone LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()
if(NOT DEFINED CGLIB_ROOT)
    set(CGLIB_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
endif()
list(APPEND CMAKE_MODULE_PATH ${CGLIB_ROOT}/cmake)
include(CGLibCommon)
include(PhantomGTest)
include(PhantomCoreLibs)
phantom_find_gtest()
if(PHANTOM_GTEST_FOUND)
    enable_testing()
endif()

# Math (Phantom::Math)
phantom_add_math_core()
file(GLOB MATHTEST_SOURCES CONFIGURE_DEPENDS ${CGLIB_ROOT}/Math/MathTest/*.cpp)
list(FILTER MATHTEST_SOURCES EXCLUDE REGEX "pch\\.cpp$")
cglib_add_test(MathTest SOURCES ${MATHTEST_SOURCES} INCLUDES ${CGLIB_ROOT}/Math/MathTest LINK MathCore)

# Every public Math header must compile on its own (no reliance on include order or a
# precompiled header). One generated TU per header; nothing is linked or run.
file(GLOB _math_headers CONFIGURE_DEPENDS RELATIVE ${CGLIB_ROOT}/Math/Math ${CGLIB_ROOT}/Math/Math/*.h)
list(FILTER _math_headers EXCLUDE REGEX "^(pch|framework)[.]h$")
set(_math_header_tus)
foreach(_h IN LISTS _math_headers)
    set(_tu ${CMAKE_CURRENT_BINARY_DIR}/MathHeaderCheck/${_h}.cpp)
    set(MATH_HEADER ${_h})
    configure_file(${CGLIB_ROOT}/cmake/MathHeaderCheck.cpp.in ${_tu} @ONLY)
    list(APPEND _math_header_tus ${_tu})
endforeach()
add_library(MathHeaderCheck OBJECT ${_math_header_tus})
target_link_libraries(MathHeaderCheck PRIVATE MathCore)
set_target_properties(MathHeaderCheck PROPERTIES DISABLE_PRECOMPILE_HEADERS ON)
target_compile_options(MathHeaderCheck PRIVATE ${PHANTOM_WARN_FLAGS})
