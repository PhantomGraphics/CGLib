cmake_minimum_required(VERSION 3.20)
project(TextureStandalone LANGUAGES CXX)
if(NOT DEFINED CGLIB_ROOT)
    get_filename_component(CGLIB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
endif()
list(APPEND CMAKE_MODULE_PATH ${CGLIB_ROOT}/cmake)
include(CGLibCommon)
include(PhantomGTest)
add_library(TextureCore STATIC Texture.cpp)
find_package(Threads REQUIRED)
target_link_libraries(TextureCore PUBLIC Threads::Threads)
target_compile_features(TextureCore PUBLIC cxx_std_20)
target_include_directories(TextureCore PUBLIC ${REPO_ROOT} ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(TextureCore PRIVATE ${CGLIB_ROOT}/ThirdParty)
target_compile_options(TextureCore PRIVATE ${PHANTOM_WARN_FLAGS})
if(MSVC)
    target_compile_definitions(TextureCore PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_library(CGLib::Texture ALIAS TextureCore)
add_executable(TextureCLI main.cpp)
target_link_libraries(TextureCLI PRIVATE TextureCore)
phantom_find_gtest()
if(PHANTOM_GTEST_FOUND)
    enable_testing()
    cglib_add_test(TextureTest SOURCES TextureTest.cpp LINK TextureCore)
endif()
